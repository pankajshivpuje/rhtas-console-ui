package policy.slsa_provenance

import rego.v1

# Allowed builder identity
allowed_builder := "https://tekton.myorg.com/chains/v2"

# Allowed source repos
allowed_source_pattern := "^github\\.com/myorg/.*$"

# Deny if builder doesn't match
deny contains msg if {
  prov := input.attestations[_]
  prov.predicate.builder.id != allowed_builder
  msg := sprintf(
    "builder %s is not allowed",
    [prov.predicate.builder.id]
  )
}

# Deny if not hermetic
deny contains msg if {
  prov := input.attestations[_]
  not prov.predicate.buildDefinition
    .buildType == "hermetic"
  msg := "build is not hermetic"
}
